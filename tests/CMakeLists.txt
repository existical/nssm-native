# Compile the production code with test-only wrappers for the entry point and
# SCM/registry calls and allocations. No replacement APIs enter the shipped executable.
include(CheckCXXSourceCompiles)
set(CMAKE_REQUIRED_INCLUDES "${PROJECT_SOURCE_DIR}/src")
set(event_compile_prefix "#define UNICODE\n#define _UNICODE\n#include <windows.h>\n#include <tchar.h>\n#include <cstdio>\n#include \"event.h\"\nvoid log_event_strings(unsigned short, unsigned long, unsigned short, const TCHAR * const *) {}\n")
check_cxx_source_compiles("${event_compile_prefix}int main() { log_event(1, 2); log_event(1, 2, L\"text\"); }" NSSM_EVENT_STRINGS_COMPILE)
check_cxx_source_compiles("${event_compile_prefix}int main() { int code = 3; log_event(1, 2, code); }" NSSM_EVENT_INTEGER_COMPILES)
if(NOT NSSM_EVENT_STRINGS_COMPILE OR NSSM_EVENT_INTEGER_COMPILES)
  message(FATAL_ERROR "Event insertion type checking failed")
endif()
unset(CMAKE_REQUIRED_INCLUDES)
unset(event_compile_prefix)
set(regression_sources ${nssm_sources})
list(REMOVE_ITEM regression_sources src/nssm.cpp src/env.cpp src/service.cpp src/registry.cpp src/settings.cpp src/hook.cpp src/io.cpp src/process.cpp src/gui.cpp src/event.cpp src/account.cpp)
list(TRANSFORM regression_sources PREPEND "${PROJECT_SOURCE_DIR}/")
add_executable(nssm_regressions ${regression_sources}
  main.cpp nssm_entry.cpp environment.cpp hooks.cpp startup.cpp registry.cpp quoting.cpp dump_quoting.cpp parsing.cpp hook_lifecycle.cpp launch.cpp logger.cpp rotation.cpp process_tree.cpp process_live.cpp install.cpp install_gui.cpp token.cpp error_messages.cpp accounts.cpp dependencies.cpp)
add_dependencies(nssm_regressions nssm_messages nssm_version)
foreach(property INCLUDE_DIRECTORIES COMPILE_DEFINITIONS COMPILE_OPTIONS LINK_LIBRARIES)
  get_target_property(value nssm ${property})
  set_property(TARGET nssm_regressions PROPERTY ${property} "${value}")
endforeach()
target_compile_features(nssm_regressions PRIVATE cxx_std_11)
target_compile_options(nssm_regressions PRIVATE /EHsc)
foreach(group environment hooks startup registry multisz quoting numeric affinity hook_lifecycle launch hook_live logger rotation process_identity process_live admin setting_ownership waits install_ownership debug_token error_messages stdin log_bytes event_insertions exit_export exit_fallback export_access account_lookup account_rights gui_retry gui_password gui_hooks dependencies control_waits)
  add_test(NAME nssm.${group} COMMAND nssm_regressions ${group})
  set_tests_properties(nssm.${group} PROPERTIES TIMEOUT 20)
endforeach()
set_tests_properties(nssm.launch nssm.hook_live nssm.logger nssm.process_live PROPERTIES TIMEOUT 60)
add_test(NAME nssm.version_identity COMMAND "${NSSM_POWERSHELL}" -NoProfile -ExecutionPolicy Bypass
  -File "${CMAKE_CURRENT_SOURCE_DIR}/version.ps1" -SourceDirectory "${PROJECT_SOURCE_DIR}" -WorkDirectory "${CMAKE_CURRENT_BINARY_DIR}")
set_tests_properties(nssm.version_identity PROPERTIES TIMEOUT 60)
add_test(NAME nssm.package_rejection COMMAND "${NSSM_POWERSHELL}" -NoProfile -ExecutionPolicy Bypass
  -File "${CMAKE_CURRENT_SOURCE_DIR}/package-rejection.ps1" -SourceDirectory "${PROJECT_SOURCE_DIR}" -WorkDirectory "${CMAKE_CURRENT_BINARY_DIR}")
set_tests_properties(nssm.package_rejection PROPERTIES TIMEOUT 60)

# disabled; controls_live starts only its own native test services.

set(NSSM_LEGACY_EXECUTABLE "" CACHE FILEPATH "Optional pinned legacy rebuild for the disposable upgrade/rollback test")
add_executable(nssm_export_live export_live.cpp)
target_compile_features(nssm_export_live PRIVATE cxx_std_11)
target_compile_options(nssm_export_live PRIVATE /W3 /EHsc)
target_link_libraries(nssm_export_live PRIVATE advapi32)
if(NSSM_ENABLE_SCM_TESTS)
  if(NSSM_LEGACY_EXECUTABLE)

      message(FATAL_ERROR "NSSM_LEGACY_EXECUTABLE does not exist; run tests/build-legacy.ps1 first.")

    add_test(NAME nssm.upgrade_live COMMAND nssm_export_live "$<TARGET_FILE:nssm>" upgrade
      "${NSSM_LEGACY_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/upgrade-worker.ps1")

  endif()
  add_test(NAME nssm.export_live COMMAND nssm_export_live "$<TARGET_FILE:nssm>")
  add_test(NAME nssm.dependencies_live COMMAND nssm_export_live "$<TARGET_FILE:nssm>" dependencies)
  add_test(NAME nssm.controls_live COMMAND nssm_export_live "$<TARGET_FILE:nssm>" controls)
  add_test(NAME nssm.lifecycle_live COMMAND nssm_export_live "$<TARGET_FILE:nssm>" lifecycle)
  set_tests_properties(nssm.lifecycle_live PROPERTIES TIMEOUT 120)
  set_tests_properties(nssm.controls_live PROPERTIES TIMEOUT 120)
  set_tests_properties(nssm.export_live PROPERTIES TIMEOUT 120)
  set_tests_properties(nssm.dependencies_live PROPERTIES TIMEOUT 120)
endif()
